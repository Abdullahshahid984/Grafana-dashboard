$wsId   = az monitor diagnostic-settings list --resource $apimId --query "value[0].workspaceId" -o tsv
"Workspace: $wsId"
$wsGuid = az monitor log-analytics workspace show --ids $wsId --query customerId -o tsv

$q = @'
AzureActivity
| where TimeGenerated > ago(60d)
| where ResourceProviderValue =~ "MICROSOFT.APIMANAGEMENT"
| where _ResourceId has "apim-ads-cus-entbusops-prd-001"
| where OperationNameValue matches regex @"(?i)(POLICIES|POLICYFRAGMENTS|NAMEDVALUES)/(WRITE|DELETE)$"
| project TimeGenerated, OperationNameValue, Caller, ActivityStatusValue
| order by TimeGenerated asc
'@
az monitor log-analytics query -w $wsGuid --analytics-query $q -o table
