```powershell
Run-LA @'
let s = datetime(2026-11-02T18:00:00Z);
let e = datetime(2026-11-02T21:00:00Z);
let apim = ApiManagementGatewayLogs
| where TimeGenerated between (s .. e)
| where BackendUrl has "api-rewards-v1"
| summarize apim_calls=count(), apim_p50=percentile(BackendTime,50), apim_p95=percentile(BackendTime,95), apim_p99=percentile(BackendTime,99) by bin(TimeGenerated, 5m);
let app = AppRequests
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards"
| summarize app_calls=count(), app_p50=percentile(DurationMs,50), app_p95=percentile(DurationMs,95), app_p99=percentile(DurationMs,99) by bin(TimeGenerated, 5m);
apim
| join kind=inner app on TimeGenerated
| project TimeGenerated, apim_calls, app_calls, gap_p50 = round(apim_p50 - app_p50, 0), gap_p95 = round(apim_p95 - app_p95, 0), gap_p99 = round(apim_p99 - app_p99, 0)
| order by TimeGenerated asc
'@
```

Run-LA @'
let s = datetime(2026-11-04T18:00:00Z);
let e = datetime(2026-11-04T21:00:00Z);
let slow = AppRequests
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards" and Name in ("POST /certificate/balance", "POST /certificate/transactions")
| where DurationMs > 1000
| project OperationId, Name, req_ms = DurationMs;
AppDependencies
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards"
| summarize dep_ms = sum(DurationMs), deps = count() by OperationId
| join kind=inner slow on OperationId
| extend in_app_ms = req_ms - dep_ms
| summarize requests = count(), p50_req = round(percentile(req_ms,50),0), p50_deps = round(percentile(dep_ms,50),0), p50_in_app = round(percentile(in_app_ms,50),0), avg_deps = round(avg(deps),1) by Name
'@
