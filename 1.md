$searchPattern = "profilebatchservice-api-config"
$secretPattern = "secretProviders"

if ($decompressed -like "*$searchPattern*" -and $decompressed -like "*$secretPattern*") {
    # Check if they're close together (within 500 chars)
    $configIndex = $decompressed.IndexOf($searchPattern)
    $secretIndex = $decompressed.IndexOf($secretPattern)
    
    if ($configIndex -ge 0 -and $secretIndex -gt $configIndex -and ($secretIndex - $configIndex) -lt 500) {
        Write-Host "SMOKING GUN: secretProviders nested in ConfigMap" -ForegroundColor Red
    } else {
        Write-Host "GOOD: secretProviders NOT nested in ConfigMap" -ForegroundColor Green
    }
} else {
    Write-Host "Could not find both patterns in output" -ForegroundColor Yellow
}
