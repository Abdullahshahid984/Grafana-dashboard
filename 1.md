```
$ErrorActionPreference = "Stop"
$code = "feat/backup.tf-update"
$f1   = "schema/instance-conf.yaml"
$f2   = "bfhaksplatform_instances/ihub-dev-eus2/conf.yaml"
$msg  = "chore(dev): add snapshot resource group setting to schema and snapshot-enabled backup instance to dev conf"

function Invoke-Git { & git @args; if ($LASTEXITCODE -ne 0) { throw "git $($args -join ' ') failed" } }

Invoke-Git fetch origin
Invoke-Git checkout dev-updates

$dirty    = git status --porcelain -- $f1 $f2                        # uncommitted edits in the working folder
$onBranch = git diff --name-only origin/main "origin/$code" -- $f1 $f2   # already committed on the code branch

if ($dirty) {
    Invoke-Git add -- $f1 $f2
    Invoke-Git commit -m $msg
} elseif ($onBranch) {
    Invoke-Git checkout "origin/$code" -- $f1 $f2
    Invoke-Git commit -m $msg
} else {
    throw "The schema and dev conf edits are not in the working folder or on $code. Restore them first (VS Code Timeline), then run this again."
}
Invoke-Git push origin dev-updates

# if they were also committed on the code branch, take them out of it (they are safe on dev-updates now)
if ($onBranch) {
    Invoke-Git checkout $code
    Invoke-Git checkout origin/main -- $f1 $f2
    Invoke-Git commit -m "chore: move schema and dev conf changes to dev-updates"
    Invoke-Git push origin $code
}

"--- $code vs main (should be only backup.tf):"
git diff --name-only origin/main "origin/$code"
"--- dev-updates vs main (should be only schema and dev conf):"
git diff --name-only origin/main origin/dev-updates
```
