function Get-ApimChanges($from, $to) {
  (az monitor activity-log list --resource-id $apimId --start-time $from --end-time $to --max-events 5000 -o json | ConvertFrom-Json) |
    Where-Object { $_.operationName.value -match '(POLICIES|POLICYFRAGMENTS|NAMEDVALUES)/(WRITE|DELETE)$' } |
    Select-Object @{n='time';e={$_.eventTimestamp}}, @{n='op';e={$_.operationName.value}}, @{n='by';e={$_.caller}}, @{n='status';e={$_.status.value}}
}
Get-ApimChanges "2026-11-01T00:00:00Z" "2026-11-09T23:59:59Z" | Sort-Object time | Format-Table -AutoSize
