```
# Get the node resource group
$nodeRG = az aks show -g <resource-group> -n <cluster-name> --query nodeResourceGroup -o tsv

# List all NSG rules
az network nsg rule list -g $nodeRG --nsg-name <nsg-name> -o json | ConvertFrom-Json | Where-Object { $_.destinationPortRange -like "*53*" -or $_.destinationPortRange -eq "*" }
```
