```
Run-LA @'
let s = datetime(2026-11-04T18:00:00Z);
let e = datetime(2026-11-04T21:00:00Z);
let apim = ApiManagementGatewayLogs
| where TimeGenerated between (s .. e)
| where _ResourceId has "prd-ext-apim-001" and ApiId in~ ("digitalapirewards", "RewardsAPI")
| summarize apim_calls=count(), apim_backend_p50=percentile(BackendTime,50), apim_backend_p95=percentile(BackendTime,95) by bin(TimeGenerated, 5m);
let app = AppRequests
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards"
| summarize app_calls=count(), app_p50=percentile(DurationMs,50), app_p95=percentile(DurationMs,95) by bin(TimeGenerated, 5m);
apim
| join kind=inner app on TimeGenerated
| extend gap_p50 = round(apim_backend_p50 - app_p50, 0), gap_p95 = round(apim_backend_p95 - app_p95, 0)
| project TimeGenerated, apim_calls, app_calls, apim_backend_p50, app_p50=round(app_p50,0), gap_p50, apim_backend_p95, app_p95=round(app_p95,0), gap_p95
| order by TimeGenerated asc
'@
```

```
Run-LA @'
let s = datetime(2026-11-04T18:00:00Z);
let e = datetime(2026-11-04T21:00:00Z);
let dep = AppDependencies
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards" and Target has "entapi-partner"
| summarize dep_calls=count(), dep_p50=percentile(DurationMs,50), dep_p95=percentile(DurationMs,95) by bin(TimeGenerated, 5m);
let papim = ApiManagementGatewayLogs
| where TimeGenerated between (s .. e)
| where _ResourceId has "partner-001" and ProductId == "rewards-certificates"
| summarize papim_calls=count(), papim_p50=percentile(TotalTime,50), papim_p95=percentile(TotalTime,95) by bin(TimeGenerated, 5m);
dep
| join kind=inner papim on TimeGenerated
| extend gap_p50 = round(dep_p50 - papim_p50, 0), gap_p95 = round(dep_p95 - papim_p95, 0)
| project TimeGenerated, dep_calls, papim_calls, dep_p50=round(dep_p50,0), papim_p50, gap_p50, dep_p95=round(dep_p95,0), papim_p95, gap_p95
| order by TimeGenerated asc
'@
```
