```powershell
Run-LA @'
let s = datetime(2026-11-04T18:00:00Z);
let e = datetime(2026-11-04T21:00:00Z);
let slow = AppRequests
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards" and Name in ("POST /certificate/balance", "POST /certificate/transactions")
| where DurationMs > 1000
| project OperationId, Name, req_ms = DurationMs;
AppDependencies
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards"
| summarize longest_dep_ms = max(DurationMs), deps = count() by OperationId
| join kind=inner slow on OperationId
| extend not_explained_ms = req_ms - longest_dep_ms
| summarize requests = count(), p50_req = round(percentile(req_ms,50),0), p50_longest_dep = round(percentile(longest_dep_ms,50),0), p50_not_explained = round(percentile(not_explained_ms,50),0) by Name
'@
```

```powershell
Run-LA @'
let s = datetime(2026-11-04T18:00:00Z);
let e = datetime(2026-11-04T21:00:00Z);
let slowOps = AppRequests
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards" and Name in ("POST /certificate/balance", "POST /certificate/transactions")
| where DurationMs > 1000
| distinct OperationId;
AppDependencies
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards" and OperationId in (slowOps)
| summarize calls = count(), p50 = round(percentile(DurationMs,50),0), p95 = round(percentile(DurationMs,95),0) by Type, Target
| order by calls desc
| take 15
'@
```
