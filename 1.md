```
$q1 = @'
ApiManagementGatewayLogs
| where TimeGenerated between (datetime(2026-11-01) .. datetime(2026-11-10))
| summarize requests=count(),
            p50_ms=percentile(TotalTime - BackendTime, 50),
            p95_ms=percentile(TotalTime - BackendTime, 95),
            p99_ms=percentile(TotalTime - BackendTime, 99),
            over_8s=countif((TotalTime - BackendTime) > 8000),
            err_5xx=countif(ResponseCode >= 500),
            policy_errors=countif(isnotempty(LastErrorMessage))
            by day=bin(TimeGenerated, 1d)
| order by day asc
'@
az monitor log-analytics query -w $wsGuid --analytics-query $q1 -o table
```
