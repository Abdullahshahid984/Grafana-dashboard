```
ApiManagementGatewayLogs
| where TimeGenerated between (datetime(2026-11-06) .. datetime(2026-11-08))
| summarize requests=count(),
            p95_apim_ms=percentile(TotalTime - BackendTime, 95),
            p95_backend_ms=percentile(BackendTime, 95),
            over_8s=countif((TotalTime - BackendTime) > 8000),
            err_5xx=countif(ResponseCode >= 500)
            by hour=bin(TimeGenerated, 1h)
| order by hour asc
```

AzureActivity
| where TimeGenerated between (datetime(2026-11-05) .. datetime(2026-11-09))
| where ResourceProviderValue =~ "MICROSOFT.APIMANAGEMENT"
| where _ResourceId has "apim-ads-cus-entbusops-prd-001"
| where OperationNameValue matches regex @"(?i)(POLICIES|POLICYFRAGMENTS|NAMEDVALUES)/(WRITE|DELETE)$"
| project TimeGenerated, OperationNameValue, Caller, ActivityStatusValue
| order by TimeGenerated asc

ApiManagementGatewayLogs
| where TimeGenerated between (datetime(2026-11-06) .. datetime(2026-11-08))
| extend size_bucket = case(RequestSize < 10000, "1 under 10KB",
                            RequestSize < 131072, "2 10KB to 128KB",
                            RequestSize < 1000000, "3 128KB to 1MB",
                            "4 over 1MB")
| summarize requests=count(), p95_apim_ms=percentile(TotalTime - BackendTime, 95),
            over_8s=countif((TotalTime - BackendTime) > 8000)
            by size_bucket
| order by size_bucket asc
