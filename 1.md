#!/usr/bin/env bash
set -euo pipefail

create_snapshot_rg() {
  local sub="$1" cluster_rg="$2" snap_rg="$3" identity="$4"

  # same tags as the existing cluster resource group
  local tags
  tags=$(az group show --name "$cluster_rg" --subscription "$sub" --query tags -o json)

  az group create --name "$snap_rg" --location eastus2 --subscription "$sub" --tags "$tags" -o none

  local scope="/subscriptions/$sub/resourceGroups/$snap_rg"
  for role in "Contributor" "Role Based Access Control Administrator"; do
    az role assignment create --assignee-object-id "$identity" \
      --assignee-principal-type ServicePrincipal --role "$role" --scope "$scope" -o none
  done

  echo "Created $snap_rg:"
  az role assignment list --scope "$scope" --query "[].{role:roleDefinitionName, principal:principalId}" -o table
}

# DEV
