function Invoke-LA([string]$query, [string]$timespan = "P7D") {
  $body = @{ query = $query; timespan = $timespan } | ConvertTo-Json -Compress
  $file = Join-Path $env:TEMP "la-body.json"
  [System.IO.File]::WriteAllText($file, $body)
  $r = az rest --method post --url "https://api.loganalytics.io/v1/workspaces/$wsGuid/query" --resource "https://api.loganalytics.io" --headers "Content-Type=application/json" --body "@$file" | ConvertFrom-Json
  $cols = $r.tables[0].columns.name
  foreach ($row in $r.tables[0].rows) { $o = [ordered]@{}; for ($i=0; $i -lt $cols.Count; $i++) { $o[$cols[$i]] = $row[$i] }; [pscustomobject]$o }
}

$hits = @()
$t = [datetime]::SpecifyKind([datetime]"2026-11-06T00:00:00", 'Utc')
$end = [datetime]::SpecifyKind([datetime]"2026-11-08T00:00:00", 'Utc')
while ($t -lt $end) {
  $from = $t.ToString("yyyy-MM-ddTHH:mm:ssZ"); $to = $t.AddMinutes(59).AddSeconds(59).ToString("yyyy-MM-ddTHH:mm:ssZ")
  $ev = az monitor activity-log list --resource-id $apimId --start-time $from --end-time $to --max-events 5000 -o json | ConvertFrom-Json
  if ($ev.Count -ge 5000) { "WARNING: 5000-event cap hit for hour starting $from" }
  $hits += $ev | Where-Object { $_.operationName.value -match '(POLICIES|POLICYFRAGMENTS|NAMEDVALUES)/(WRITE|DELETE)$' }
  $t = $t.AddHours(1)
}
$hits | Select-Object @{n='time';e={$_.eventTimestamp}}, @{n='op';e={$_.operationName.value}}, @{n='by';e={$_.caller}}, @{n='status';e={$_.status.value}} | Sort-Object time | Format-Table -AutoSize
