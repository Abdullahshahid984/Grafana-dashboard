```
$ev = (az monitor activity-log list --resource-id $apimId --start-time $start --max-events 1000 -o json | ConvertFrom-Json) |
      Where-Object { $_.operationName.value -match 'policies|policyFragments|namedValues' }
"Matching events: $($ev.Count)"

"--- per day ---"
$ev | Group-Object { ([datetime]$_.eventTimestamp).ToString('yyyy-MM-dd') } | Sort-Object Name | Select-Object Name, Count | Format-Table -AutoSize

"--- all events ---"
$ev | Select-Object @{n='time';e={$_.eventTimestamp}}, @{n='op';e={$_.operationName.value}}, @{n='by';e={$_.caller}}, @{n='status';e={$_.status.value}} |
      Sort-Object time | Format-Table -AutoSize
```
