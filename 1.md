```
Run-LA @'
let s = datetime(2026-11-02T18:30:00Z);
let e = datetime(2026-11-02T20:30:00Z);
let pods = KubePodInventory
| where TimeGenerated between (s .. e) and Namespace == "api-rewards-v1"
| distinct PodUid, Name, PodCreationTimeStamp;
Perf
| where TimeGenerated between (s .. e)
| where ObjectName == "K8SContainer" and CounterName in ("cpuUsageNanoCores", "memoryWorkingSetBytes")
| where InstanceName contains "apirewards"
| extend PodUid = tostring(split(InstanceName, "/")[-2])
| join kind=inner pods on PodUid
| summarize avg_val = round(avg(CounterValue) / 1e6, 0), p95_val = round(percentile(CounterValue, 95) / 1e6, 0) by Name, PodCreationTimeStamp, CounterName
| order by CounterName asc, PodCreationTimeStamp asc
'@
```
