```
Run-LA @'
let inc = datetime(2026-11-02T18:30:00Z);
let base = datetime(2026-10-26T17:30:00Z);
ApiManagementGatewayLogs
| where TimeGenerated between (inc .. inc + 2h) or TimeGenerated between (base .. base + 2h)
| where ApiId has "reward"
| extend Instance = extract(@"service/([^/]+)$", 1, tolower(_ResourceId))
| extend Day = iff(TimeGenerated >= inc, "Nov02", "Oct26")
| extend GatewayTime = TotalTime - BackendTime
| summarize calls=count(), p50=round(percentile(TotalTime,50),0), p95=round(percentile(TotalTime,95),0),
    backend_p95=round(percentile(BackendTime,95),0), gateway_p95=round(percentile(GatewayTime,95),0)
    by Instance, ApiId, Day
| order by Instance asc, ApiId asc, Day asc
'@
```
