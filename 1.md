```
# --- setup ---
$apim = "apim-ads-cus-entbusops-prd-001"
az account set --subscription "cf7e12f8-ce8c-43c8-a1c0-1287d71499ca"
$rg     = az apim list --query "[?name=='$apim'].resourceGroup | [0]" -o tsv
$apimId = az apim show -g $rg -n $apim --query id -o tsv
$api    = "2022-08-01"

# 1. Size, network mode, private IPs
"=== 1. APIM basics ==="
az apim show -g $rg -n $apim --query "{sku:sku.name,units:sku.capacity,vnetType:virtualNetworkType,privateIps:privateIpAddresses}" -o json

# 2. Is Traceable in the All APIs policy, and what timeout / error settings?
"=== 2. Traceable in All APIs policy ==="
az rest --method get --url "https://management.azure.com$apimId/policies/policy?api-version=$api&format=rawxml" --query properties.value -o tsv > allapis-policy.xml
Select-String -Path allapis-policy.xml -Pattern "traceable|ext_cap|send-request|send-one-way|timeout|ignore-error" -Context 0,1

# 3. The Traceable named value (agent IP and port)
"=== 3. Named value ==="
az apim nv list -g $rg --service-name $apim --query "[?contains(name,'traceable') || contains(name,'tpa')].{name:name,secret:secret,value:value}" -o table

# 4. When did the policy / named values change (last 60 days)?
"=== 4. Changes in 60 days ==="
$start = (Get-Date).AddDays(-60).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")
az monitor activity-log list --resource-id $apimId --start-time $start `
  --query "[?contains(operationName.value,'policies') || contains(operationName.value,'policyFragments') || contains(operationName.value,'namedValues')].{time:eventTimestamp,op:operationName.value,by:caller,status:status.value}" -o table

# 5. Does the agent VM exist and is it running?
"=== 5. Agent VM ==="
az vm list -d --query "[?contains(name,'raceable') || contains(name,'tpa')].{name:name,rg:resourceGroup,state:powerState,ip:privateIps}" -o table
```
