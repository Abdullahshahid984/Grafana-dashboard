# Create the role for reading configmaps
kubectl create role api-transactions-configmap-reader -n api-transactions-v1 --verb=get --verb=list --resource=configmaps

# Bind it to the service account
kubectl create rolebinding api-transactions-configmap-reader -n api-transactions-v1 --role=api-transactions-configmap-reader --serviceaccount=api-transactions-v1:sa-api-transactions-v1
