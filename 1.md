function Run-LA($q) {
  @{ query = $q } | ConvertTo-Json -Compress | Out-File -Encoding ascii q.json
  $r = az rest --method post --url "https://management.azure.com/subscriptions/8a6206c2-20f1-4b1a-8050-a2edb2ebc5da/resourceGroups/rg-ads-cus-core-prd-mon-001/providers/Microsoft.OperationalInsights/workspaces/law-ads-cus-core-prd-001/api/query?api-version=2017-01-01-preview" --body "@q.json" --query "Tables[0]" -o json | ConvertFrom-Json
  $cols = $r.Columns.ColumnName
  $r.Rows | ForEach-Object { $row = $_; $o = [ordered]@{}; for ($i = 0; $i -lt $cols.Count; $i++) { $o[$cols[$i]] = $row[$i] }; [pscustomobject]$o } | Format-Table -AutoSize
}
