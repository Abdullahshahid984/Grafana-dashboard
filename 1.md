```
Write-Host "Checking Global 'list secrets' access..." -ForegroundColor Cyan
kubectl get namespaces -o jsonpath='{.items[*].metadata.name}' | ForEach-Object {
    $ns = $_
    $canI = kubectl auth can-i list secrets --namespace $ns --as="12d1b154-8a86-4e5c-bee3-1c428f2d2490" --as-group="system:authenticated" --as-group="system:masters"
    [PSCustomObject]@{
        Namespace  = $ns
        ListSecretsAllowed = $canI
    }
} | Format-Table -AutoSize
```
