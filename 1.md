```
$ErrorActionPreference = "Stop"

# ---- fill in the SIT identity object id (leave as is to skip SIT) ----
$sitPipelineObjectId = "<SIT pipeline identity object id>"
# -----------------------------------------------------------------------

$envs = @(
  @{ Name = "dev"; Sub = "1fd4eaa2-e0f5-4e47-a395-12bad3a0cd73"; Cluster = "rg-bfhaks-ihub-eus2-dev-01"; Snap = "rg-bfhaks-ihub-eus2-dev-01-snapshots"; Identity = "fa6e5c41-9977-4989-a38b-b55ca66e1644" },
  @{ Name = "sit"; Sub = "850e7f17-9c38-4461-a988-fd26995f1bc1"; Cluster = "rg-bfhaks-ihub-eus2-sit-01"; Snap = "rg-bfhaks-ihub-eus2-sit-01-snapshots"; Identity = $sitPipelineObjectId }
)

function Invoke-Az { & az @args; if ($LASTEXITCODE -ne 0) { throw "az $($args -join ' ') failed" } }

foreach ($e in $envs) {
  if ($e.Identity -like "<*") { "Skipping $($e.Name): identity object id not set"; continue }
  "=== $($e.Name) ==="

  $tags = az group show --name $e.Cluster --subscription $e.Sub --query "tags" -o json | ConvertFrom-Json
  $t    = $tags.PSObject.Properties | ForEach-Object { "$($_.Name)=$($_.Value)" }

  Invoke-Az group create --name $e.Snap --location eastus2 --subscription $e.Sub --tags $t -o none

  $scope = "/subscriptions/$($e.Sub)/resourceGroups/$($e.Snap)"
  foreach ($role in @("Contributor", "Role Based Access Control Administrator")) {
    Invoke-Az role assignment create --assignee-object-id $e.Identity --assignee-principal-type ServicePrincipal --role $role --scope $scope -o none
  }

  "Created $($e.Snap) and assigned roles. Check:"
  az role assignment list --scope $scope --query "[].{role:roleDefinitionName, principal:principalId}" -o table
}
```
