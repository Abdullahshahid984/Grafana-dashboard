```
Run-LA @'
let s = datetime(2026-10-01T00:00:00Z);
let e = datetime(2026-11-03T04:00:00Z);
let pods = KubePodInventory
| where TimeGenerated between (s .. e) and Namespace == "api-rewards-v1"
| distinct PodUid, Name;
Perf
| where TimeGenerated between (s .. e)
| where ObjectName == "K8SContainer" and CounterName == "cpuUsageNanoCores"
| where InstanceName contains "apirewards"
| where hourofday(TimeGenerated) between (17 .. 20) and dayofweek(TimeGenerated) between (1d .. 5d)
| extend PodUid = tostring(split(InstanceName, "/")[-2])
| join kind=inner pods on PodUid
| summarize avg_mcores = round(avg(CounterValue) / 1e6, 0) by Name, Week = startofweek(TimeGenerated)
| order by Name asc, Week asc
'@
```
