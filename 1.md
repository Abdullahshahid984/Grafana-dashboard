# Get all workload MIs and check which ones have poc-02 federated credential
$namespaces = kubectl get sa -A -o json | ConvertFrom-Json

$namespaces.items | Where-Object {
    $_.metadata.annotations.'azure.workload.identity/client-id'
} | ForEach-Object {
    $ns = $_.metadata.namespace
    $sa = $_.metadata.name
    $clientId = $_.metadata.annotations.'azure.workload.identity/client-id'
    
    $miName = az identity list --query "[?clientId=='$clientId'].name" -o tsv
    
    $hasPoc02 = az identity federated-credential list `
        --identity-name $miName `
        --resource-group rg-bfhaks-ihub_workloads-eus2-dev-01 `
        --query "[?contains(name,'poc02') || contains(issuer,'efebe195')].name" `
        -o tsv

    if ($hasPoc02) {
        Write-Host "ALREADY HAS POC-02: $ns/$sa"
    } else {
        Write-Host "MISSING POC-02: $ns/$sa"
    }
}
