$secretData = (kubectl get secret sh.helm.release.v1.api-accountlookupexperience.v32 -n api-accountlookupexperience-v1 -o jsonpath="{.data.release}")
$base64Decoded1 = [System.Convert]::FromBase64String($secretData)
$base64String1 = [System.Text.Encoding]::UTF8.GetString($base64Decoded1)
$base64Decoded2 = [System.Convert]::FromBase64String($base64String1)

$ms = New-Object System.IO.MemoryStream (,$base64Decoded2)
$gzip = New-Object System.IO.Compression.GZipStream ($ms, [System.IO.Compression.CompressionMode]::Decompress)
$reader = New-Object System.IO.StreamReader ($gzip)
$unzippedManifest = $reader.ReadToEnd()

# This checks if the broken string exists inside the configuration map
$unzippedManifest | Select-String -Pattern "secretProviders" -Context 5, 10
