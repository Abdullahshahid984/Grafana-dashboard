Find the load balancer
```
$body = @{ query = "Resources | where properties contains '10.37.136.100' | project name, type, resourceGroup, subscriptionId" } | ConvertTo-Json -Compress
[System.IO.File]::WriteAllText("$env:TEMP\arg.json", $body)
$lb = az rest --method post --url "https://management.azure.com/providers/Microsoft.ResourceGraph/resources?api-version=2021-03-01" --body "@$env:TEMP\arg.json" --query "data[0]" -o json | ConvertFrom-Json
"LB: $($lb.name) in $($lb.resourceGroup)"
```
Read its configuration: ports, health probe, backends
```
function Cnt($x) { if ($null -eq $x) { 0 } else { @($x).Count } }

$lbj = az network lb show --subscription $lb.subscriptionId -g $lb.resourceGroup -n $lb.name -o json | ConvertFrom-Json
"SKU: $($lbj.sku.name)   Location: $($lbj.location)"

"--- Front end ---"
$lbj.frontendIPConfigurations | Select-Object name, privateIPAddress, privateIPAllocationMethod | Format-Table -AutoSize

"--- Rules (which ports are forwarded) ---"
$lbj.loadBalancingRules | Select-Object name, protocol, frontendPort, backendPort, idleTimeoutInMinutes | Format-Table -AutoSize

"--- Health probes ---"
$lbj.probes | Select-Object name, protocol, port, requestPath, intervalInSeconds, numberOfProbes | Format-Table -AutoSize

"--- Backend pools ---"
foreach ($p in $lbj.backendAddressPools) {
  "{0}: {1} network-interface members, {2} IP members" -f $p.name, (Cnt $p.backendIPConfigurations), (Cnt $p.loadBalancerBackendAddresses)
}

```

Was it healthy on Friday?

```
$m = az monitor metrics list --resource $lbj.id --metrics DipAvailability VipAvailability --aggregation Average --interval PT1H `
     --start-time 2026-11-06T00:00:00Z --end-time 2026-11-07T00:00:00Z -o json | ConvertFrom-Json
foreach ($mv in $m.value) {
  $d = $mv.timeseries[0].data
  "{0}: lowest hourly value {1}; hours below 100: {2}" -f $mv.name.value, ($d | Measure-Object average -Minimum).Minimum, @($d | Where-Object { $_.average -lt 100 }).Count
  $d | Where-Object { $_.average -lt 100 } | Select-Object timeStamp, average | Format-Table -AutoSize
}
```
