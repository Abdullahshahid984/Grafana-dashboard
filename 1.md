```powershell
Run-LA @'
let s = datetime(2026-11-02T18:00:00Z);
let e = datetime(2026-11-02T21:00:00Z);
let dep = AppDependencies
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards" and Target has "entapi-partner"
| summarize dep_calls=count(), dep_p50=percentile(DurationMs,50), dep_p95=percentile(DurationMs,95), dep_p99=percentile(DurationMs,99), dep_max=max(DurationMs) by bin(TimeGenerated, 5m);
let papim = ApiManagementGatewayLogs
| where TimeGenerated between (s .. e)
| where _ResourceId has "partner-001" and ProductId == "rewards-certificates"
| summarize papim_calls=count(), papim_p50=percentile(TotalTime,50), papim_p95=percentile(TotalTime,95), papim_p99=percentile(TotalTime,99), papim_max=max(TotalTime) by bin(TimeGenerated, 5m);
dep
| join kind=inner papim on TimeGenerated
| project TimeGenerated, dep_calls, papim_calls, gap_p50=round(dep_p50-papim_p50,0), gap_p95=round(dep_p95-papim_p95,0), gap_p99=round(dep_p99-papim_p99,0), dep_max=round(dep_max,0), papim_max
| order by TimeGenerated asc
'@
```

```powershell
Run-LA @'
let d2 = datetime(2026-11-02T18:00:00Z);
let d4 = datetime(2026-11-04T18:00:00Z);
let apim = ApiManagementGatewayLogs
| where TimeGenerated between (d2 .. d2 + 3h) or TimeGenerated between (d4 .. d4 + 3h)
| where BackendUrl has "HOSTHERE"
| extend Day = iff(TimeGenerated >= d4, "Nov04", "Nov02")
| summarize apim_calls=count(), apim_p50=percentile(BackendTime,50), apim_p95=percentile(BackendTime,95) by Day;
let app = AppRequests
| where TimeGenerated between (d2 .. d2 + 3h) or TimeGenerated between (d4 .. d4 + 3h)
| where AppRoleName == "ROLEHERE"
| extend Day = iff(TimeGenerated >= d4, "Nov04", "Nov02")
| summarize app_calls=count(), app_p50=percentile(DurationMs,50), app_p95=percentile(DurationMs,95) by Day;
apim
| join kind=inner app on Day
| project Day, apim_calls, app_calls, gap_p50=round(apim_p50-app_p50,0), gap_p95=round(apim_p95-app_p95,0)
'@
```

```powershell
| summarize joined = count(), over_150ms = countif(outside_app_ms > 150), over_500ms = countif(outside_app_ms > 500), p99 = round(percentile(outside_app_ms,99),0), max_ms = round(max(outside_app_ms),0)
```
