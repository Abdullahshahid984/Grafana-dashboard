function Invoke-LA2([string]$query, [string]$timespan = "P7D") {
  $body = @{ query = $query; timespan = $timespan } | ConvertTo-Json -Compress
  $file = Join-Path $env:TEMP "la-body.json"
  [System.IO.File]::WriteAllText($file, $body)
  $r = az rest --method post --url "https://management.azure.com$($wsId)/api/query?api-version=2017-01-01-preview" --headers "Content-Type=application/json" --body "@$file" | ConvertFrom-Json
  $t = $r.tables[0]
  $cols = $t.columns | ForEach-Object { if ($_.name) { $_.name } else { $_.ColumnName } }
  foreach ($row in $t.rows) { $o = [ordered]@{}; for ($i=0; $i -lt $cols.Count; $i++) { $o[$cols[$i]] = $row[$i] }; [pscustomobject]$o }
}
