ApiManagementGatewayLogs
| where TimeGenerated between (datetime(2026-11-06T04:20:00Z) .. datetime(2026-11-06T17:40:00Z))
| where LastErrorSource == "send-request"
| summarize errors=count(), first_seen=min(TimeGenerated), last_seen=max(TimeGenerated),
            p50_total_ms=percentile(TotalTime, 50), max_total_ms=max(TotalTime)
            by LastErrorReason, LastErrorScope, LastErrorSection, ApiId
| order by errors desc
