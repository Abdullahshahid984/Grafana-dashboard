
```powershell
Run-LA @'
ApiManagementGatewayLogs
| where TimeGenerated between (datetime(2026-11-04T18:00:00Z) .. datetime(2026-11-04T21:00:00Z))
| where BackendUrl has "api-rewards-v1"
| extend Instance = extract(@"service/([^/]+)$", 1, tolower(_ResourceId))
| summarize calls = count() by Instance, ApiId
| order by calls desc
'@
```

```powershell

AppRequests
| where TimeGenerated between (datetime(2026-11-04T18:00:00Z) .. datetime(2026-11-04T21:00:00Z))
| where AppRoleName == "apirewards"
| summarize calls = count(), p50 = round(percentile(DurationMs,50),0), p95 = round(percentile(DurationMs,95),0) by Name
| order by calls desc
| take 20
'@
```


```powershell
Run-LA @'
let s = datetime(2026-11-04T18:00:00Z);
let e = datetime(2026-11-04T21:00:00Z);
let apim = ApiManagementGatewayLogs
| where TimeGenerated between (s .. e)
| where BackendUrl has "api-rewards-v1"
| summarize apim_calls=count(), apim_p50=percentile(BackendTime,50), apim_p95=percentile(BackendTime,95), apim_p99=percentile(BackendTime,99) by bin(TimeGenerated, 5m);
let app = AppRequests
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards"
| summarize app_calls=count(), app_p50=percentile(DurationMs,50), app_p95=percentile(DurationMs,95), app_p99=percentile(DurationMs,99) by bin(TimeGenerated, 5m);
apim
| join kind=inner app on TimeGenerated
| project TimeGenerated, apim_calls, app_calls,
    gap_p50 = round(apim_p50 - app_p50, 0), gap_p95 = round(apim_p95 - app_p95, 0), gap_p99 = round(apim_p99 - app_p99, 0)
| order by TimeGenerated asc
'@
```

```powershell
dep
| join kind=inner papim on TimeGenerated
| project TimeGenerated, dep_calls, papim_calls, dep_p99=round(dep_p99,0), papim_p99, gap_p99=round(dep_p99 - papim_p99,0), dep_max=round(dep_max,0), papim_max
| order by TimeGenerated asc
'@
```
