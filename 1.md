```
KubeEvents
| where TimeGenerated between (datetime(2026-11-06T15:00:00Z) .. datetime(2026-11-06T17:30:00Z))
| where Namespace == "api-statements-v1"
| where Reason in ("Unhealthy", "Killing", "BackOff", "Failed", "OOMKilling")
| project TimeGenerated, Name, Reason, Message
| order by TimeGenerated asc
```

```
KubePodInventory
| where TimeGenerated between (datetime(2026-11-06T14:00:00Z) .. datetime(2026-11-06T18:00:00Z))
| where Namespace == "api-statements-v1"
| summarize restarts=max(ContainerRestartCount), status=any(PodStatus) by Name, bin(TimeGenerated, 5m)
| order by TimeGenerated asc
```
