```powershell
Run-LA @'
let s = datetime(2026-11-02T19:00:00Z);
let e = datetime(2026-11-02T20:00:00Z);
let apimDep = AppDependencies
| where TimeGenerated between (s .. e)
| where AppRoleName startswith "apim-ads-cus-entbusops-prd-001" and Target has "api-rewards-v1"
| project Id, apim_backend_ms = DurationMs;
AppRequests
| where TimeGenerated between (s .. e)
| where AppRoleName == "apirewards"
| project ParentId, AppRoleInstance, app_ms = DurationMs
| join kind=inner apimDep on $left.ParentId == $right.Id
| extend gap_ms = apim_backend_ms - app_ms
| summarize pairs = count(), p50 = round(percentile(gap_ms,50),0), p95 = round(percentile(gap_ms,95),0), p99 = round(percentile(gap_ms,99),0), over_150ms = countif(gap_ms > 150), over_500ms = countif(gap_ms > 500) by AppRoleInstance
| order by AppRoleInstance asc
'@
```
